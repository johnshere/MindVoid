## 1. 浮窗渲染（Windows）

### 1.1 创建浮窗

```rust
use windows::Win32::UI::WindowsAndMessaging::*;
use windows::Win32::Graphics::Dwm::*;

fn create_overlay_window() -> Result<HWND> {
    let hwnd = unsafe {
        CreateWindowExW(
            WS_EX_LAYERED           // 分层窗口（透明）
            | WS_EX_TRANSPARENT     // 点击穿透
            | WS_EX_TOPMOST         // 始终置顶
            | WS_EX_NOACTIVATE      // 不激活（无焦点）
            | WS_EX_TOOLWINDOW,     // 不显示在任务栏
            wc.lpszClassName,
            "MindVoid",
            WS_POPUP,               // 无边框
            0, 0, 400, 60,
            None, None, hinstance, None,
        )
    };

    // 启用毛玻璃效果（Windows 10+ Acrylic / Windows 11 Mica）
    let accent = DWM_BLURBEHIND {
        dwFlags: DWM_BB_ENABLE | DWM_BB_BLURREGION,
        fEnable: true.into(),
        hRgnBlur: CreateRectRgn(0, 0, 400, 60),
        fTransitionOnMaximized: false.into(),
    };
    unsafe { DwmEnableBlurBehindWindow(hwnd, &accent) };

    // 设置透明度
    unsafe { SetLayeredWindowAttributes(hwnd, 0, 217, LWA_ALPHA) }; // 85% 不透明

    Ok(hwnd)
}
```

### 1.2 候选列表渲染

候选项最多 5 个，用 GDI+ 绘制：

```rust
fn render_candidates(hwnd: HWND, predictions: &[Prediction], selected: usize) {
    let mut ps = PAINTSTRUCT::default();
    let hdc = unsafe { BeginPaint(hwnd, &mut ps) };
    let graphics = Graphics::from_hdc(hdc);  // 创建 GDI+ Graphics 对象

    // 背景
    let bg = SolidBrush::new(Color::from_argb(40, 0, 0, 0)); // 半透明黑底
    graphics.fill_rectangle(&bg, &Rect::new(0, 0, 400, 60));

    // 候选词（水平排列）
    let font = Font::new(L"Microsoft YaHei", 14.0).unwrap();
    for (i, pred) in predictions.iter().enumerate() {
        let x = 10 + i as i32 * 80;
        let color = if i == selected { Color::white() } else { Color::from_argb(180, 255, 255, 255) };

        // 序号标签 (Alt+1~5)
        let num_brush = SolidBrush::new(Color::gray());
        graphics.draw_string(&format!("&{}", i + 1), &font, &PointF::new(x as f32, 5.0), &num_brush);
        // 候选文本
        let text_brush = SolidBrush::new(color);
        graphics.draw_string(&pred.text, &font, &PointF::new((x + 20) as f32, 5.0), &text_brush);
    }

    // GDI+ 对象在离开作用域时自动释放 (Drop)
    unsafe { EndPaint(hwnd, &ps) };
}
```

### 1.3 跟随光标

```rust
fn update_overlay_position(hwnd: HWND) {
    // 使用 GetGUIThreadInfo 获取准确的插入符位置
    let mut gui_info = GUITHREADINFO::default();
    gui_info.cbSize = std::mem::size_of::<GUITHREADINFO>() as u32;
    unsafe { GetGUIThreadInfo(0, &mut gui_info) };

    let mut pos = gui_info.rcCaret;
    unsafe { ClientToScreen(gui_info.hwndCaret, &mut pos as *mut RECT as *mut POINT) };

    let (mut x, mut y) = (pos.left + 10, pos.bottom + 5); // 右下偏移

    // 屏幕边缘翻转
    let screen_w = unsafe { GetSystemMetrics(SM_CXSCREEN) };
    let screen_h = unsafe { GetSystemMetrics(SM_CYSCREEN) };

    let w = 400; let h = 60;
    if x + w > screen_w { x = screen_w - w - 10; }
    if y + h > screen_h { y = gui_info.rcCaret.top - h - 5; } // 翻到光标上方
    if x < 0 { x = 10; }
    if y < 0 { y = 10; }

    unsafe { SetWindowPos(hwnd, HWND_TOPMOST, x, y, w, h, SWP_NOACTIVATE | SWP_SHOWWINDOW) };
}
```

## 2. 交互处理

### 2.1 鼠标点击选词

```rust
fn on_mouse_click(hwnd: HWND, x: i32, _y: i32) {
    let idx = x / 80; // 每个候选占 80px 宽
    if idx < predictions.len() {
        on_candidate_selected(idx);
    }
}
```

### 2.2 全局组合键选词

```rust
fn on_hotkey(hwnd: HWND, hotkey_id: i32) {
    match hotkey_id {
        HK_ALT_1..=HK_ALT_5 => {
            let idx = hotkey_id - HK_ALT_1;
            on_candidate_selected(idx as usize);
        }
        HK_TOGGLE => {
            is_visible = !is_visible;
            if is_visible {
                unsafe { ShowWindow(hwnd, SW_SHOWNA) };
            } else {
                unsafe { ShowWindow(hwnd, SW_HIDE) };
            }
        }
        _ => {}
    }
}
```

组合键注册：

```rust
// Alt+1 ~ Alt+5
```

### 2.3 候选选择动作

```rust
fn on_candidate_selected(idx: usize) {
    if let Some(pred) = predictions.get(idx) {
        // 1. 模拟键盘输入选中的候选词（替换用户正在打的文本）
        simulate_text_input(&pred.text);

        // 2. 通知训练分流器：用户选择了候选
        training_pipeline.push_sample(TrainSample {
            context: ring_buffer.snapshot().to_string(),
            output: pred.text.clone(),
            confidence: Confidence::High,
        });

        // 3. 隐藏浮窗，等待下一次预测
        hide_overlay();
    }
}
```

**模拟文本输入：** 使用 `SendInput` API 模拟键盘输入候选词，或如果环境文本可用，精确替换 IME 组合字符。

## 3. 消息循环与钩子协作

```
┌─────────────────────────────────────┐
│           UI 主线程消息循环           │
│                                     │
│  while GetMessage(&msg) {           │
│    match msg.message {              │
│      WM_HOTKEY   → on_hotkey()     │
│      WM_LBUTTONUP → on_click()     │
│      WM_PAINT    → render()        │
│      WM_TIMER    → update_position()│
│      WM_APP_PREDICT → show_pred()  │  ← 推理结果回调（PostMessage）
│    }                                │
│  }                                  │
└─────────────────────────────────────┘
         ↑ PostMessage (线程安全)
┌─────────────────────────────────────┐
│        推理线程 (独立)               │
│                                     │
│  result = model.predict(features);  │
│  PostMessage(ui_hwnd, WM_APP_PREDICT, │
│              result_ptr, 0);        │
└─────────────────────────────────────┘
```

**跨线程通知：** 推理线程通过 `PostMessage` 将预测结果发到 UI 线程。UI 线程在消息循环中接收并渲染。无需额外锁——Windows 消息队列天然线程安全。

## 4. 应用管理

### 4.1 系统托盘

```rust
fn create_tray_icon(hwnd: HWND) {
    let mut nid = NOTIFYICONDATAW {
        cbSize: std::mem::size_of::<NOTIFYICONDATAW>() as u32,
        hWnd: hwnd,
        uID: 1,
        uFlags: NIF_ICON | NIF_MESSAGE | NIF_TIP,
```

### 4.2 开机自启

```rust
use winreg::enums::*;
use winreg::RegKey;

fn set_autostart(enabled: bool) {
    let hkcu = RegKey::predef(HKEY_CURRENT_USER);
    let path = r"Software\Microsoft\Windows\CurrentVersion\Run";

    if enabled {
        let (key, _) = hkcu.create_subkey(path).unwrap();
        key.set_value("MindVoid", &std::env::current_exe().unwrap().to_str().unwrap()).unwrap();
    } else {
        if let Ok(key) = hkcu.open_subkey_with_flags(path, KEY_WRITE) {
            key.delete_value("MindVoid").ok();
        }
    }
}
```

## 6. 安装与分发

| 项 | 方案 |
|----|------|
| 打包格式 | 便携 zip（无 MSI，降低复杂度） |
| 文件结构 | `mindvoid.exe` + `config.toml`（首次运行自动生成） + `mindvoid_models/` |
| 开机自启 | 注册表 `HKCU\...\Run` 写入 exe 路径 |
| 辅助功能权限引导 | 首次启动检测 UIA 可用性，失败则弹窗图文引导 |
| 更新策略 | Phase 2：检查 GitHub Release → 下载 zip → 覆盖替换 |

### 权限引导流程

```
启动 → 尝试 UIA 获取焦点控件
  ├─ 成功 → 正常启动
  └─ 失败(权限) → 弹窗：
       "需要辅助功能权限以读取输入框文本"
       [打开 Windows 设置] [忽略（功能受限）]
```
